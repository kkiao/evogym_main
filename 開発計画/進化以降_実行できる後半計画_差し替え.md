## 1. 先に `reinforce.py` を完成形へ置き換える

`train_fixed.py` と `evolve.py` の両方が、同じ「体を学習させる」「方策を評価する」処理を使えるようにする。これをしないと、進化側に学習コードを丸ごと複製することになり、直す場所が二つになってしまう。

**編集するファイル：** `src/reinforce.py`  
**すること：** ファイルの中身を次のコードへ置き換える。

```python
import gymnasium as gym
import evogym.envs
import torch
from torch.distributions import Normal

from src.policy import Policy


def calculate_returns(rewards, gamma):
    """各時刻から先にもらう報酬の合計 G_t を計算する。"""
    returns = []
    future_return = 0.0

    for reward in reversed(rewards):
        future_return = reward + gamma * future_return
        returns.insert(0, future_return)

    return torch.tensor(returns, dtype=torch.float32)


def make_mean_action(policy, obs, action_low, action_high):
    """ランダムさを入れない、評価・動画用の行動を作る。"""
    obs_tensor = torch.tensor(obs, dtype=torch.float32).unsqueeze(0)
    mean = policy(obs_tensor)

    low = torch.tensor(action_low, dtype=torch.float32)
    high = torch.tensor(action_high, dtype=torch.float32)
    center = (low + high) / 2
    half_range = (high - low) / 2

    mean_in_range = center + half_range * torch.tanh(mean)
    action = torch.max(torch.min(mean_in_range, high), low)

    return action.squeeze(0).detach().numpy()


def choose_training_action(policy, obs, action_low, action_high):
    """学習用に、平均の周辺からランダムな行動とlog_probを作る。"""
    obs_tensor = torch.tensor(obs, dtype=torch.float32).unsqueeze(0)
    mean = policy(obs_tensor)

    low = torch.tensor(action_low, dtype=torch.float32)
    high = torch.tensor(action_high, dtype=torch.float32)
    center = (low + high) / 2
    half_range = (high - low) / 2
    mean_in_range = center + half_range * torch.tanh(mean)

    std = torch.ones_like(mean_in_range) * 0.20
    distribution = Normal(mean_in_range, std)
    raw_action = distribution.sample()
    action = torch.max(torch.min(raw_action, high), low)
    log_prob = distribution.log_prob(raw_action).sum()

    return action.squeeze(0).detach().numpy(), log_prob


def update_policy(optimizer, log_probs, returns):
    """REINFORCEの損失を作り、Policyの重みを一回更新する。"""
    log_probs_tensor = torch.stack(log_probs)
    normalized_returns = (returns - returns.mean()) / (returns.std() + 0.000001)
    loss = -(log_probs_tensor * normalized_returns).sum()

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    return loss.item()


def run_training_episode(env, policy, optimizer, gamma, max_steps):
    """1エピソードを動かし、最後にREINFORCEで一回だけ更新する。"""
    obs, _ = env.reset()
    rewards = []
    log_probs = []
    episode_return = 0.0

    for step_number in range(max_steps):
        action, log_prob = choose_training_action(
            policy,
            obs,
            env.action_space.low,
            env.action_space.high,
        )
        next_obs, reward, terminated, truncated, _ = env.step(action)

        rewards.append(reward)
        log_probs.append(log_prob)
        episode_return = episode_return + reward
        obs = next_obs

        if terminated or truncated:
            break

    returns = calculate_returns(rewards, gamma)
    loss = update_policy(optimizer, log_probs, returns)
    return episode_return, loss


def train_body(task_name, body, episodes, max_steps, seed=7):
    """指定された体へ、新しいPolicyを作って指定回数だけ学習させる。"""
    torch.manual_seed(seed)
    env = gym.make(task_name, body=body, render_mode=None)
    first_obs, _ = env.reset(seed=seed)

    observation_size = len(first_obs)
    action_size = env.action_space.shape[0]
    policy = Policy(observation_size, action_size)
    optimizer = torch.optim.Adam(policy.parameters(), lr=0.0003)
    training_returns = []

    for episode_number in range(episodes):
        episode_return, loss = run_training_episode(
            env,
            policy,
            optimizer,
            gamma=0.99,
            max_steps=max_steps,
        )
        training_returns.append(episode_return)
        print("train", episode_number, "return", episode_return, "loss", loss)

    env.close()
    return policy, training_returns


def evaluate_body(task_name, body, policy, episodes, max_steps, seed=100):
    """重みを変えず、平均行動だけで方策の実力を測る。"""
    env = gym.make(task_name, body=body, render_mode=None)
    evaluation_returns = []

    for episode_number in range(episodes):
        obs, _ = env.reset(seed=seed + episode_number)
        episode_return = 0.0

        for step_number in range(max_steps):
            with torch.no_grad():
                action = make_mean_action(
                    policy,
                    obs,
                    env.action_space.low,
                    env.action_space.high,
                )

            next_obs, reward, terminated, truncated, _ = env.step(action)
            episode_return = episode_return + reward
            obs = next_obs

            if terminated or truncated:
                break

        evaluation_returns.append(episode_return)

    env.close()
    return sum(evaluation_returns) / len(evaluation_returns)
```

### この置換で増えた関数の役割

| 関数 | すること | どこから呼ばれるか |
| --- | --- | --- |
| `make_mean_action` | ランダムさなしの行動を作る | 評価、GIF |
| `choose_training_action` | ランダムな学習用行動と`log_prob`を作る | 学習エピソード |
| `run_training_episode` | 1回動かし、エピソードの最後に1回更新する | `train_body` |
| `train_body` | 新しいPolicyを作り、指定回数学習する | 固定体学習、進化の各個体 |
| `evaluate_body` | 重みを変えず平均行動だけで平均報酬を測る | 進化のfitness、最終比較 |

`with torch.no_grad():` は「評価中は学習しないので、重みを変えるための計算履歴を記録しない」という指定である。評価が速くなり、誤って学習しない。

## 2. 固定体のCSVを実際に作る `train_fixed.py`

前に書いた直接の学習ループは使わず、上の `train_body` を呼ぶ。固定体と最良体を同じファイルで学習できる。

**編集するファイル：** `src/train_fixed.py`  
**すること：** 中身を次へ置き換える。

```python
import csv
from pathlib import Path

import numpy as np
import torch

from src.body import make_fixed_body
from src.reinforce import evaluate_body, train_body


# 固定体を学習するときの設定。
TASK_NAME = "Climber-v0"
EPISODES = 60
MAX_STEPS = 500
OUTPUT_PREFIX = "fixed"
BODY_FILE = None  # 最良体を再学習するときだけ "results/best_body.npy" に変える。


def main():
    results_dir = Path("results")
    results_dir.mkdir(exist_ok=True)

    if BODY_FILE is None:
        body = make_fixed_body()
    else:
        body = np.load(BODY_FILE)

    policy, training_returns = train_body(
        TASK_NAME,
        body,
        episodes=EPISODES,
        max_steps=MAX_STEPS,
    )
    evaluation_return = evaluate_body(
        TASK_NAME,
        body,
        policy,
        episodes=3,
        max_steps=MAX_STEPS,
    )

    np.save(results_dir / f"{OUTPUT_PREFIX}_body.npy", body)
    torch.save(policy.state_dict(), results_dir / f"{OUTPUT_PREFIX}_policy.pt")

    with open(results_dir / f"{OUTPUT_PREFIX}_training.csv", "w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        writer.writerow(["episode", "train_return"])
        for episode_number, episode_return in enumerate(training_returns):
            writer.writerow([episode_number, episode_return])

    with open(results_dir / f"{OUTPUT_PREFIX}_evaluation.txt", "w", encoding="utf-8") as file:
        file.write(f"mean_evaluation_return={evaluation_return}\n")

    print("平均評価報酬:", evaluation_return)


if __name__ == "__main__":
    main()
```

**実行：**

```powershell
python -m src.train_fixed
```

`Path("results")` はresultsフォルダの場所を扱うためのオブジェクト、`mkdir(exist_ok=True)` はフォルダがなければ作り、既にあってもエラーにしないメソッドである。最後の3行は「このファイルを直接実行したときだけ`main()`を呼ぶ」というPythonの定型文である。

この実行で必ず次ができる。

```text
results/fixed_body.npy
results/fixed_policy.pt
results/fixed_training.csv
results/fixed_evaluation.txt
```

## 3. グラフを実際に出す `plot_results.py`

**新規作成：** `src/plot_results.py`

```python
import csv
from pathlib import Path

import matplotlib.pyplot as plt


def read_csv_rows(path):
    with open(path, newline="", encoding="utf-8") as file:
        return list(csv.DictReader(file))


def plot_training(csv_path, output_path, title):
    rows = read_csv_rows(csv_path)
    episodes = [int(row["episode"]) for row in rows]
    returns = [float(row["train_return"]) for row in rows]

    plt.figure(figsize=(8, 4))
    plt.plot(episodes, returns, label="train return")
    plt.xlabel("episode")
    plt.ylabel("total reward")
    plt.title(title)
    plt.grid()
    plt.legend()
    plt.tight_layout()
    plt.savefig(output_path)
    plt.close()


def plot_evolution(csv_path, output_path):
    rows = read_csv_rows(csv_path)
    grouped = {}

    for row in rows:
        generation = int(row["generation"])
        fitness = float(row["fitness"])
        if generation not in grouped:
            grouped[generation] = []
        grouped[generation].append(fitness)

    generations = sorted(grouped.keys())
    best_values = [max(grouped[generation]) for generation in generations]
    mean_values = [sum(grouped[generation]) / len(grouped[generation]) for generation in generations]

    plt.figure(figsize=(8, 4))
    plt.plot(generations, best_values, marker="o", label="best fitness")
    plt.plot(generations, mean_values, marker="o", label="mean fitness")
    plt.xlabel("generation")
    plt.ylabel("fitness")
    plt.title("Evolution result")
    plt.grid()
    plt.legend()
    plt.tight_layout()
    plt.savefig(output_path)
    plt.close()


def main():
    results_dir = Path("results")

    if (results_dir / "fixed_training.csv").exists():
        plot_training(
            results_dir / "fixed_training.csv",
            results_dir / "fixed_learning_curve.png",
            "Fixed body training",
        )

    if (results_dir / "best_training.csv").exists():
        plot_training(
            results_dir / "best_training.csv",
            results_dir / "best_learning_curve.png",
            "Best body training",
        )

    if (results_dir / "evolution_history.csv").exists():
        plot_evolution(
            results_dir / "evolution_history.csv",
            results_dir / "evolution_curve.png",
        )


if __name__ == "__main__":
    main()
```

**実行：**

```powershell
python -m src.plot_results
```

`csv.DictReader` はCSVの1行目の列名を使い、`row["episode"]` のように読み込めるようにする。`plt.plot` が折れ線、`plt.savefig` がPNG保存、`plt.close` がそのグラフのメモリを閉じる処理である。

## 4. GIFを実際に出す `render_demo.py`

**新規作成：** `src/render_demo.py`

```python
from pathlib import Path

import gymnasium as gym
import evogym.envs
import imageio.v2 as imageio
import numpy as np
import torch

from src.policy import Policy
from src.reinforce import make_mean_action


# 実行ごとにこの4行だけを変更する。
TASK_NAME = "Climber-v0"
BODY_FILE = "results/fixed_body.npy"
POLICY_FILE = "results/fixed_policy.pt"  # 学習前GIFなら None にする。
OUTPUT_FILE = "results/fixed_after.gif"
MAX_STEPS = 300


def main():
    body = np.load(BODY_FILE)
    env = gym.make(TASK_NAME, body=body, render_mode="rgb_array")
    obs, _ = env.reset(seed=100)

    policy = Policy(len(obs), env.action_space.shape[0])
    if POLICY_FILE is not None:
        policy.load_state_dict(torch.load(POLICY_FILE, map_location="cpu"))

    frames = []
    for step_number in range(MAX_STEPS):
        frame = env.render()
        if frame is not None:
            frames.append(frame)

        with torch.no_grad():
            action = make_mean_action(
                policy,
                obs,
                env.action_space.low,
                env.action_space.high,
            )

        next_obs, reward, terminated, truncated, _ = env.step(action)
        obs = next_obs

        if terminated or truncated:
            break

    env.close()
    imageio.mimsave(OUTPUT_FILE, frames, fps=30)
    print("GIFを保存しました:", OUTPUT_FILE)


if __name__ == "__main__":
    main()
```

1. `POLICY_FILE = None`、`OUTPUT_FILE = "results/fixed_before.gif"` にして実行

## 5. 実行できる `evolve.py` を作る

このファイルが、進化実験の入口である。各体へ `train_body` を呼び、`evaluate_body` でfitnessを測り、CSV・体ファイルを保存し、次世代を作る。

**新規作成：** `src/evolve.py`

```python
import csv
from pathlib import Path
import random

import numpy as np

from src.body import make_fixed_body, mutate_body
from src.reinforce import evaluate_body, train_body


TASK_NAME = "Climber-v0"
POPULATION_SIZE = 4
GENERATIONS = 3
SHORT_TRAIN_EPISODES = 15
EVALUATION_EPISODES = 2
MAX_STEPS = 500
SEED = 7


def make_initial_population():
    parent = make_fixed_body()
    population = [parent]

    while len(population) < POPULATION_SIZE:
        population.append(mutate_body(parent))

    return population


def choose_best(scored_bodies):
    # 各要素は (fitness, body) の二つ組。fitnessの高い順に並べる。
    scored_bodies.sort(key=lambda item: item[0], reverse=True)
    return scored_bodies[0]


def make_next_generation(best_body):
    next_generation = [best_body.copy()]

    while len(next_generation) < POPULATION_SIZE:
        next_generation.append(mutate_body(best_body))

    return next_generation


def main():
    random.seed(SEED)
    results_dir = Path("results")
    results_dir.mkdir(exist_ok=True)

    population = make_initial_population()
    history_rows = []
    best_fitness_ever = float("-inf")
    best_body_ever = None

    for generation in range(GENERATIONS):
        print("=== generation", generation, "===")
        scored_bodies = []

        for individual, body in enumerate(population):
            print("individual", individual, "を学習・評価します")

            # 各体には新しいPolicyと同じ学習回数を与える。
            policy, training_returns = train_body(
                TASK_NAME,
                body,
                episodes=SHORT_TRAIN_EPISODES,
                max_steps=MAX_STEPS,
                seed=SEED,
            )
            fitness = evaluate_body(
                TASK_NAME,
                body,
                policy,
                episodes=EVALUATION_EPISODES,
                max_steps=MAX_STEPS,
            )

            body_file = results_dir / f"g{generation}_i{individual}_body.npy"
            np.save(body_file, body)
            history_rows.append([
                generation,
                individual,
                fitness,
                training_returns[-1],
                str(body_file),
            ])
            scored_bodies.append((fitness, body))

            if fitness > best_fitness_ever:
                best_fitness_ever = fitness
                best_body_ever = body.copy()

            print("fitness:", fitness)

        generation_best_fitness, generation_best_body = choose_best(scored_bodies)
        print("世代の最高fitness:", generation_best_fitness)

        # 最終世代の後は子を作る必要がない。
        if generation < GENERATIONS - 1:
            population = make_next_generation(generation_best_body)

    np.save(results_dir / "best_body.npy", best_body_ever)

    with open(results_dir / "evolution_history.csv", "w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        writer.writerow([
            "generation",
            "individual",
            "fitness",
            "last_train_return",
            "body_file",
        ])
        writer.writerows(history_rows)

    with open(results_dir / "best_fitness.txt", "w", encoding="utf-8") as file:
        file.write(str(best_fitness_ever))

    print("全世代での最高fitness:", best_fitness_ever)
    print("最良の体を保存しました: results/best_body.npy")


if __name__ == "__main__":
    main()
```

### `evolve.py` が実際にする順番

```text
main()
  → 固定体 + 突然変異3体を作る
  → 各体を15エピソード学習する（毎回、新しいPolicy）
  → 2エピソードを学習せずに評価し、平均報酬をfitnessにする
  → 体を .npy として保存、CSVの1行を追加する
  → fitness最大の体を選ぶ
  → その体を残し、3回突然変異して次世代の4体にする
  → 3世代後、全個体中の最良体をbest_body.npyへ保存する
```
